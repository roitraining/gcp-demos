import asyncio, sys
sys.path.insert(0, "examples")
from _common import ask, bootstrap, flush_spans, install_console_spans
bootstrap()
from google.adk.runners import InMemoryRunner
from demo_agent.agent import root_agent

async def main():
    install_console_spans()
    runner = InMemoryRunner(agent=root_agent, app_name="tracing_1")
    session = await runner.session_service.create_session(app_name="tracing_1", user_id="u1")
    for _ in range(5):
        await ask(runner, "What's the weather in London?", session_id=session.id)
    flush_spans()

asyncio.run(main())
PY
